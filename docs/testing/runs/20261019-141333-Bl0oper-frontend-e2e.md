```js
// F3 dry run of the SCRUM-44 (E05-S05) click-through on the local real stack.
// Usage: node f3_44.mjs <repo root> <scratchpad>
// Needs: freshly reset and seeded connectsphere_dev_stack in container
// cs-scrum34-test, API on 3001, Vite on 5173.
import { createRequire } from 'node:module';
import { execSync } from 'node:child_process';
import { mkdirSync } from 'node:fs';
import { join } from 'node:path';

const [repo, scratch] = process.argv.slice(2);
const require = createRequire(join(repo, 'package.json'));
const { chromium } = require('@playwright/test');
const base = 'http://localhost:5173';
const shots = join(scratch, 'f3');
mkdirSync(shots, { recursive: true });

const results = [];
const check = (id, name, ok, seen) => {
  results.push({ id, name, ok, seen });
  console.log(`${ok ? 'PASS' : 'FAIL'} ${id} ${name}${seen ? ` | ${seen}` : ''}`);
};
const psql = (sql) => execSync('docker exec -i cs-scrum34-test psql -U postgres -d connectsphere_dev_stack -At -F "|"', { input: sql }).toString().trim();
const stage = (name) => {
  execSync(`docker exec -i cs-scrum34-test psql -U postgres -d connectsphere_dev_stack -q -v stage=${name} -f -`, { input: require('node:fs').readFileSync(join(scratch, 'e05s05-stages.sql')) });
  console.log(`-- stage ${name}`);
};

const browser = await chromium.launch();
console.log('chromium', browser.version());
const contexts = {};
async function as(email) {
  if (!contexts[email]) {
    const context = await browser.newContext({ viewport: { width: 1280, height: 900 }, extraHTTPHeaders: { origin: base } });
    const login = await context.request.post(`${base}/api/auth/session`, { data: { email, password: 'ValidPass123' } }); // pragma: allowlist secret
    if (login.status() !== 200) throw new Error(`login ${email} ${login.status()}`);
    contexts[email] = { context, page: await context.newPage() };
  }
  return contexts[email].page;
}
async function shot(page, name) {
  await page.screenshot({ path: join(shots, `${name}-1280.png`), fullPage: false });
  await page.setViewportSize({ width: 393, height: 852 });
  await page.waitForTimeout(300);
  await page.screenshot({ path: join(shots, `${name}-393.png`), fullPage: false });
  const sideways = await page.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth);
  await page.setViewportSize({ width: 1280, height: 900 });
  return sideways;
}
const orchid = psql("SELECT id FROM venues WHERE name = 'Orchid Hall'");
const text = async (locator) => (await locator.innerText()).replace(/\s+/g, ' ').trim();
async function calendarRange(page, from, to, venueStaff = false) {
  if (venueStaff) {
    await page.goto(`${base}/venue/availability`);
    await page.getByLabel('Venue', { exact: true }).selectOption({ label: 'Orchid Hall' });
  } else {
    await page.goto(`${base}/coordinator/venues/${orchid}/calendar`);
  }
  await page.getByLabel('From', { exact: true }).fill(from);
  await page.getByLabel('To', { exact: true }).fill(to);
  await page.getByRole('button', { name: 'Show range' }).click();
  const days = page.getByRole('list', { name: /^Availability for Orchid Hall/ });
  await days.waitFor();
  await page.waitForTimeout(500);
  return text(days);
}
async function requestBooking(page, eventCode, start, end) {
  await page.goto(`${base}/coordinator/events/${eventCode}/venues/${orchid}/suitability`);
  const form = page.getByRole('form', { name: 'Request venue booking' });
  await form.waitFor();
  await page.getByLabel('Booking start').fill(start);
  await page.getByLabel('Booking end').fill(end);
  await page.getByRole('button', { name: 'Submit booking request' }).click();
  const outcome = page.getByText('Booking request submitted').or(page.locator('.ui-alert-error'));
  await outcome.first().waitFor();
  return text(page.locator('main'));
}

try {
  // TC_06: Orchid Hall at the seed's 0/0.
  stage('tc06');
  const staff = await as('venue_a@connectsphere.com');
  await staff.goto(`${base}/venue/inventory`);
  const orchidCard = staff.getByRole('article', { name: 'Orchid Hall', exact: true });
  await orchidCard.waitFor();
  const orchidFacts = await text(orchidCard);
  check('TC_E05S05_06', 'Orchid Hall reads 0 and 0', /Setup time 0 minutes/.test(orchidFacts) && /Turnaround time 0 minutes/.test(orchidFacts), orchidFacts.slice(0, 300));
  await shot(staff, 'TC_E05S05_06-1');
  const coordA = await as('coord_a@connectsphere.com');
  const october = await calendarRange(coordA, '2026-10-06', '2026-10-08');
  check('TC_E05S05_06', 'calendar 6-8 Oct shows no buffer parts', !/Setup time|Turnaround time/.test(october), october.slice(0, 600));
  await shot(coordA, 'TC_E05S05_06-2');

  // TC_01: Venue Staff set 30 and 45.
  await staff.goto(`${base}/venue/inventory/${orchid}/edit`);
  await staff.getByLabel('Setup time').fill('30');
  await staff.getByLabel('Turnaround time').fill('45');
  await shot(staff, 'TC_E05S05_01-1');
  await staff.getByRole('button', { name: 'Save changes' }).click();
  await staff.getByRole('heading', { name: 'Venue inventory' }).waitFor();
  await orchidCard.waitFor();
  const saved = await text(orchidCard);
  const lotus = await text(staff.getByRole('article', { name: 'Lotus Room', exact: true }));
  check('TC_E05S05_01', 'Orchid Hall shows 30 and 45; Lotus Room still 0 and 0',
    /Setup time 30 minutes/.test(saved) && /Turnaround time 45 minutes/.test(saved) && /Setup time 0 minutes/.test(lotus) && /Turnaround time 0 minutes/.test(lotus), `${saved.slice(0, 200)} || ${lotus.slice(0, 200)}`);
  check('TC_E05S05_01', 'database stores 30 and 45', psql(`SELECT setup_time_minutes||'/'||turnaround_time_minutes FROM venues WHERE id='${orchid}'`) === '30/45');
  await shot(staff, 'TC_E05S05_01-2');

  // TC_02: buffered window on the calendar.
  stage('s1');
  const january = await calendarRange(coordA, '2027-01-15', '2027-01-15');
  check('TC_E05S05_02', 'calendar shows setup 9:30-10:00, event 10:00-12:00, turnaround 12:00-12:45', /Setup time/.test(january) && /Turnaround time/.test(january), january.slice(0, 700));
  await shot(coordA, 'TC_E05S05_02-1');

  // TC_07: search for EVT-3001 (12:30-14:00).
  await coordA.goto(`${base}/coordinator/events/EVT-3001/venues`);
  await coordA.getByRole('button', { name: 'Search venues' }).click();
  const result = coordA.getByRole('region', { name: 'Orchid Hall', exact: true });
  await result.waitFor();
  const found = await text(result);
  check('TC_E05S05_07', 'Orchid Hall Unavailable, overlap 12:30-12:45 named',
    /Unavailable/.test(found) && found.includes('another booking needs the venue from 15 Jan 2027, 12:30 pm to 15 Jan 2027, 12:45 pm, including setup and turnaround.'), found.slice(0, 600));
  check('TC_E05S05_07', 'no Hold this venue link for Orchid Hall', (await result.getByRole('link', { name: 'Hold this venue' }).count()) === 0);
  await result.scrollIntoViewIfNeeded();
  await shot(coordA, 'TC_E05S05_07-1');

  // TC_04: Request A, then restore, then Request B.
  stage('s2');
  const coordB = await as('coord_b@connectsphere.com');
  const requestA = await requestBooking(coordB, 'EVT-2003', '2027-01-15T13:15', '2027-01-15T15:00');
  check('TC_E05S05_04', 'Request A (13:15) accepted as Pending', requestA.includes('Booking request submitted') && requestA.includes('Your request is Pending'), requestA.slice(0, 300));
  check('TC_E05S05_04', 'database: A pending', psql(`SELECT b.status FROM venue_bookings b JOIN events e ON e.id=b.event_id WHERE e.event_code='EVT-2003' AND b.venue_id='${orchid}'`) === 'pending');
  await shot(coordB, 'TC_E05S05_04-1');
  stage('s2restore');
  const requestB = await requestBooking(coordB, 'EVT-2003', '2027-01-15T13:14', '2027-01-15T15:00');
  check('TC_E05S05_04', 'Request B (13:14) refused naming 9:30 am to 12:45 pm',
    requestB.includes('This venue is unavailable from 15 Jan 2027, 9:30 am to 15 Jan 2027, 12:45 pm, including setup and turnaround.'), requestB.slice(0, 400));
  check('TC_E05S05_04', 'database: no EVT-2003 booking; EVT-101 still confirmed',
    psql(`SELECT count(*) FROM venue_bookings b JOIN events e ON e.id=b.event_id WHERE e.event_code='EVT-2003' AND b.venue_id='${orchid}'`) === '0'
    && psql(`SELECT b.status FROM venue_bookings b JOIN events e ON e.id=b.event_id WHERE e.event_code='EVT-101'`) === 'confirmed');
  await shot(coordB, 'TC_E05S05_04-2');

  // TC_05: request next to a block.
  stage('s3');
  const block = await requestBooking(coordA, 'EVT-3001', '2027-01-15T12:00', '2027-01-15T14:00');
  check('TC_E05S05_05', 'request 12:00-14:00 next to block 08:00-12:00 accepted as Pending', block.includes('Your request is Pending'), block.slice(0, 300));
  await shot(coordA, 'TC_E05S05_05-1');

  // TC_03: turnaround 0 -> 45 creates a conflict.
  stage('s4');
  await staff.goto(`${base}/venue/inventory/${orchid}/edit`);
  await staff.getByLabel('Turnaround time').fill('45');
  await staff.getByRole('button', { name: 'Save changes' }).click();
  const alert = staff.getByRole('alert');
  await alert.waitFor();
  const warning = await text(alert);
  check('TC_E05S05_03', 'save confirmation lists the pair',
    warning.includes('Changes saved, and some bookings now conflict') && warning.includes('EVT-2003 Client B Isolation Event (EVT-2003), 15 Jan 2027, 12:30 pm – 2:00 pm: conflicts with EVT-101 Client A Planning Event'), warning);
  const sideways = await shot(staff, 'TC_E05S05_03-1');
  check('TC_E05S05_03', 'no sideways page scroll at 393 px', !sideways);
  check('TC_E05S05_03', 'database: both confirmed, EVT-2003 flagged',
    psql(`SELECT string_agg(e.event_code||':'||b.status||':'||b.buffer_conflict, ',' ORDER BY e.event_code) FROM venue_bookings b JOIN events e ON e.id=b.event_id WHERE b.venue_id='${orchid}' AND b.booking_range && '[2027-01-15 00:00+08,2027-01-16 00:00+08)'`) === 'EVT-101:confirmed:false,EVT-2003:confirmed:true');
  const flagged = await calendarRange(staff, '2027-01-15', '2027-01-15', true);
  check('TC_E05S05_03', 'Venue Staff calendar marks EVT-2003 as conflicting', /EVT-2003 Client B Isolation Event[^]*Buffer conflict — needs review/.test(flagged) && !/EVT-101 Client A Planning Event · Buffer conflict/.test(flagged), flagged.slice(0, 900));
  await shot(staff, 'TC_E05S05_03-2');
  const coordBView = await calendarRange(await as('coord_b@connectsphere.com'), '2027-01-15', '2027-01-15');
  check('TC_E05S05_03', 'coord_b calendar marks own EVT-2003 as conflicting', coordBView.includes('EVT-2003 Client B Isolation Event · Buffer conflict — needs review'), coordBView.slice(0, 900));
  const notices = psql(`SELECT u.email||':'||n.title FROM notifications n JOIN users u ON u.id=n.user_id WHERE n.title='Booking conflict detected' ORDER BY u.email`);
  check('TC_E05S05_03', 'coord_a and coord_b each have one notice', notices === 'coord_a@connectsphere.com:Booking conflict detected\ncoord_b@connectsphere.com:Booking conflict detected', notices.replace(/\n/g, '; '));
  await coordB.goto(`${base}/notifications`);
  await coordB.getByText('Booking conflict detected').first().click();
  await coordB.waitForTimeout(500);
  check('TC_E05S05_03', 'coord_b reads the notice', (await text(coordB.locator('main'))).includes('creating a scheduling conflict for "EVT-2003 Client B Isolation Event"'));
  await shot(coordB, 'TC_E05S05_03-3');
} catch (error) {
  check('SCRIPT', 'stopped', false, String(error).slice(0, 400));
} finally {
  await browser.close();
  const failed = results.filter((r) => !r.ok).length;
  console.log(`${results.length - failed}/${results.length} passed`);
}
```
