## Script

`measure_periods.mjs`, kept outside the repository; reproduced in full so the
method can be checked and run again. The password is the seed's public test
password (`tests/seed/SEED_DATA.md`). Its `pragma: allowlist secret` comment
was added after the run, for the repository's secret scanner; nothing else
differs from what ran.

```js
// Measures the Unavailable periods table on two equipment pages (local real stack).
// Usage: node measure_periods.mjs <repo root>
import { createRequire } from 'node:module';
import { join } from 'node:path';

const require = createRequire(join(process.argv[2], 'package.json'));
const { chromium } = require('@playwright/test');
const base = 'http://localhost:5173';

const browser = await chromium.launch();
console.log('chromium', browser.version());
const signIn = async (email) => {
  const context = await browser.newContext({ viewport: { width: 1280, height: 900 }, extraHTTPHeaders: { origin: base } });
  const login = await context.request.post(`${base}/api/auth/session`, { data: { email, password: 'ValidPass123' } }); // pragma: allowlist secret - public seed test password
  console.log('login', email, login.status());
  return context;
};
const context = await signIn('tech_a@connectsphere.com');
const api = context.request;

const list = await (await api.get(`${base}/api/equipment`)).json();
const items = list.equipment ?? list;
const mic = items.find((i) => i.name === 'Microphone-Wireless');
const proj = items.find((i) => i.name === 'Projector-HD');
// Steps 1 and 2 of the click-through script: coord_a requests 2 Projector-HD
// for EVT-3003 and tech_a reserves them, so Projector-HD has reservation history.
const requests = `${base}/api/equipment?mode=requests&event=EVT-3003`;
const coord = await signIn('coord_a@connectsphere.com');
const saved = await coord.request.post(requests, { data: { action: 'saveRequest', equipmentId: proj.id, quantity: 2, notes: '' } });
console.log('saveRequest', saved.status());
const { requestId } = await saved.json();
await coord.close();
const reserved = await api.post(requests, { data: { action: 'reserve', requestId, quantity: 2 } });
console.log('reserve', reserved.status());
// The same periods as steps 7 and 9 of the click-through script.
const marks = [
  { action: 'markUnavailable', id: mic.id, from: '2026-12-01', to: '2026-12-05', quantity: 6, reason: 'Sensor repair' },
  { action: 'markUnavailable', id: proj.id, from: '2026-10-25', to: '2026-10-25', quantity: 1, reason: 'Lamp check' },
];
for (const data of marks) {
  const mark = await api.post(`${base}/api/equipment`, { data });
  console.log('mark', data.reason, mark.status());
}

const page = await context.newPage();
for (const item of [mic, proj]) {
  await page.goto(`${base}/support/catalogue/${item.id}`);
  // The action buttons render after the table, once the page knows the caller may edit.
  await page.getByRole('table', { name: 'Unavailable periods' }).getByRole('button', { name: 'Return to service…' }).first().waitFor();
  const m = await page.evaluate(() => {
    const px = (el) => Math.round(el.getBoundingClientRect().width);
    const byCaption = (text) => [...document.querySelectorAll('table.ui-table')].find((t) => t.querySelector('caption')?.textContent === text);
    const table = byCaption('Unavailable periods');
    const wrap = table.parentElement;
    const shell = document.querySelector('.ui-page');
    const history = byCaption('Reservation history');
    const button = [...table.querySelectorAll('button')].find((b) => b.textContent.startsWith('Return to service'));
    return {
      viewport: window.innerWidth,
      pageScrolls: document.documentElement.scrollWidth > document.documentElement.clientWidth,
      pageWidth: px(shell), pageMax: getComputedStyle(shell).maxWidth,
      wrapClient: wrap.clientWidth, tableWidth: px(table), tableMinWidth: getComputedStyle(table).minWidth,
      tableScroll: wrap.scrollWidth, overflows: wrap.scrollWidth > wrap.clientWidth,
      buttonRightPastWrap: Math.round(button.getBoundingClientRect().right - wrap.getBoundingClientRect().right),
      historyTable: history ? px(history) : 'none (empty state)',
      columns: [...table.querySelectorAll('thead th')].map((th) => `${th.textContent}:${px(th)}`),
    };
  });
  console.log(item.name, JSON.stringify(m));
}
await browser.close();
```
